// handler to get messages by conversation id within a date range

func (h *MessageHandler) GetMessageByConversationIDAndDateRange(c *gin.Context) {
	id, err := uuid.Parse(c.Query("conversationid"))
	if err != nil {
		apierrors.HandleBindingError(c, err)
		return
	}
	if id == uuid.Nil {
		apierrors.HandleErrorWithCustomMessage(c, "ID cannot be null", err)
		return
	}

	fromStr := c.Query("from")
	toStr := c.Query("to")
	if fromStr == "" || toStr == "" {
		apierrors.HandleErrorWithCustomMessage(c, "from and to are required", nil)
		return
	}

	from, err := time.Parse(time.RFC3339, fromStr)
	if err != nil {
		apierrors.HandleBindingError(c, err)
		return
	}
	to, err := time.Parse(time.RFC3339, toStr)
	if err != nil {
		apierrors.HandleBindingError(c, err)
		return
	}

	if from.After(to) {
		apierrors.HandleErrorWithCustomMessage(c, "from must be before to", nil)
		return
	}
	if to.Sub(from) > 3*30*24*time.Hour {
		apierrors.HandleErrorWithCustomMessage(c, "date range must not exceed 3 months", nil)
		return
	}

	msgs, err := h.messagerepo.GetMessagesByConversationIDAndDateRange(c, id, from, to)
	if err != nil {
		apierrors.HandleDBError(c, err)
		return
	}

	resp := response.GetMessageByConversationIdApiResponse{
		StatusCodeAndMessage: port.ListSuccess,
		Data:                 msgs,
	}
	handleSuccess(c, resp)
}
